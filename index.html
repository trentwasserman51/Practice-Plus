-- Practice Plus: team accounts (coaches edit, players view) -- Run once in Supabase: SQL Editor -> New query -> paste -> Run. -- Uses the same Supabase project as Range Plus, in separate pp_ tables. create table if not exists public.pp_members ( email text primary key, role text not null check (role in ('coach','player')), added_at timestamptz not null default now() ); create table if not exists public.pp_docs ( path text primary key, data jsonb not null, updated_at timestamptz not null default now() ); alter table public.pp_members enable row level security; alter table public.pp_docs enable row level security; -- The signed-in person's role ('coach', 'player', or null if not on the team) create or replace function public.pp_role() returns text language sql stable security definer set search_path = public as $$ select role from public.pp_members where email = lower(auth.jwt() ->> 'email') $$; grant execute on function public.pp_role() to authenticated; drop policy if exists pp_members_read on public.pp_members; drop policy if exists pp_members_coach_insert on public.pp_members; drop policy if exists pp_members_coach_update on public.pp_members; drop policy if exists pp_members_coach_delete on public.pp_members; create policy pp_members_read on public.pp_members for select to authenticated using (email = lower(auth.jwt() ->> 'email') or public.pp_role() = 'coach'); create policy pp_members_coach_insert on public.pp_members for insert to authenticated with check (public.pp_role() = 'coach'); create policy pp_members_coach_update on public.pp_members for update to authenticated using (public.pp_role() = 'coach') with check (public.pp_role() = 'coach'); create policy pp_members_coach_delete on public.pp_members for delete to authenticated using (public.pp_role() = 'coach'); drop policy if exists pp_docs_read on public.pp_docs; drop policy if exists pp_docs_coach_insert on public.pp_docs; drop policy if exists pp_docs_coach_update on public.pp_docs; drop policy if exists pp_docs_coach_delete on public.pp_docs; create policy pp_docs_read on public.pp_docs for select to authenticated using (public.pp_role() is not null); create policy pp_docs_coach_insert on public.pp_docs for insert to authenticated with check (public.pp_role() = 'coach'); create policy pp_docs_coach_update on public.pp_docs for update to authenticated using (public.pp_role() = 'coach') with check (public.pp_role() = 'coach'); create policy pp_docs_coach_delete on public.pp_docs for delete to authenticated using (public.pp_role() = 'coach'); grant select, insert, update, delete on public.pp_members, public.pp_docs to authenticated;
